{{/* Ordered, non-empty library categories. Usage: `range partial "library-categories.html" .` Go templates walk a map in key order, so `hugo.Data.library` is rebuilt as a slice sorted by each file's own `weight`. The items are sorted here too, once, so this page and the homepage row cannot disagree about "newest". `sort`, not `.ByDate` — these are plain maps, not pages. A missing `date` means "in progress" and floats above everything dated. Splitting the slice first is also what keeps `sort` away from a nil key. */}} {{ $cats := slice }} {{ range $key, $cat := hugo.Data.library }} {{/* Skip categories with no entries — an empty heading is noise. */}} {{ with $cat.items }} {{ $dated := slice }} {{ $undated := slice }} {{ range . }} {{ if .date }} {{ $dated = $dated | append . }} {{ else }} {{ $undated = $undated | append . }} {{ end }} {{ end }} {{/* `date_label` doubles as the i18n suffix — `finished` renders "Finished ". `bare` is the one non-word value: date, no label. `cover_ratio` is a category setting because shape is a property of the medium, not an entry. */}} {{ $cats = $cats | append (dict "key" $key "display" ($cat.display | default "list") "weight" ($cat.weight | default 99) "dateLabel" ($cat.date_label | default "") "coverRatio" ($cat.cover_ratio | default "square") "items" ($undated | append (sort $dated "date" "desc")) ) }} {{ end }} {{ end }} {{ return sort $cats "weight" }}
