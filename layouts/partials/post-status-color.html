{{/* Usage: `partial "post-status-color.html" .` — the single place a blog post's status colour is decided. Returns a dict of literal class strings: `chip` (the flat badge) and `ramp` (the post card's gradient corner chip), so the two shapes are the same colour by construction and neither needs a safelist. The colours are the `--status-*` variables in main.css — one emerald ramp, identical in both themes. An unknown or archived status falls through to the paper chip, the one entry that *is* theme-aware. `return`, not rendered output, and that is load-bearing: a partial that ends by printing hands its caller the template's text, newlines and all. This one used to, returning "\n\nprimary" and silently uncolouring every status on the site. */}} {{ $archived := dict "chip" "badge-ghost" "ramp" "bg-linear-to-br from-base-100 to-base-200 text-base-content" }} {{ $status := dict "seed" (dict "chip" "[--badge-color:var(--status-seed-deep)] [--badge-fg:var(--status-seed-ink)]" "ramp" "bg-linear-to-br from-(--status-seed-start) to-(--status-seed-deep) text-(--status-seed-ink)") "growing" (dict "chip" "[--badge-color:var(--status-growing-deep)] [--badge-fg:var(--status-growing-ink)]" "ramp" "bg-linear-to-br from-(--status-growing-start) to-(--status-growing-deep) text-(--status-growing-ink)") "living" (dict "chip" "[--badge-color:var(--status-living-deep)] [--badge-fg:var(--status-living-ink)]" "ramp" "bg-linear-to-br from-(--status-living-start) to-(--status-living-deep) text-(--status-living-ink)") "archived" $archived }} {{ return (index $status (.Params.status | default "") | default $archived) }}
