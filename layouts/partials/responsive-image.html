{{/* Usage: `partial "responsive-image.html" (dict "page" . "src" .Params.cover "alt" .Title "class" "w-full border-4 border-base-200 shadow-md object-cover" "width" "800")` Resolves src against the page's own bundle resources first, falling back to site-wide assets. Generates a responsive picture element with WebP and JPEG fallback, width/height, and lazy loading. */}} {{ $img := "" }} {{ with .page }} {{ $img = .Resources.GetMatch $.src }} {{ end }} {{ if not $img }} {{ $img = resources.Get .src }} {{ end }} {{ if $img }} {{/* Never resize past the source's own width: `.Resize` happily upscales. */}} {{ $width := int (.width | default 800) }} {{ if lt $img.Width $width }} {{ $width = $img.Width }} {{ end }} {{ $webp := $img.Resize (printf "%dx webp" $width) }} {{ $jpeg := $img.Resize (printf "%dx jpg" $width) }} {{/* `contents` removes the picture box so the img is laid out by the parent — otherwise inside a flex/grid parent `w-full` never applies. */}} {{ end }}
