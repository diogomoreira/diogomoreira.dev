{{/* Usage: `$posts := partial "blog-posts.html" .` — every blog post in every language, newest first. The feed is deliberately mixed: EN and pt-br posts are different writing, not translations, so a language-scoped list hides half of it. `union`, not `append`: it keeps the page.Pages type, so .ByDate / .Reverse / .GroupByDate survive. One partial owns the ordering and four surfaces read it. */}} {{ $posts := where site.RegularPages "Section" "blog" }} {{ range hugo.Sites }} {{ if ne .Language.Lang site.Language.Lang }} {{ $posts = $posts | union (where .RegularPages "Section" "blog") }} {{ end }} {{ end }} {{/* A translated post would otherwise appear twice: keep the current site's copy, drop the foreign one. No such pair exists today. */}} {{ $dupes := slice }} {{ range $posts }} {{ $p := . }} {{ if ne $p.Language.Lang site.Language.Lang }} {{ range $p.Translations }} {{ if eq .Language.Lang site.Language.Lang }}{{ $dupes = $dupes | append $p }}{{ end }} {{ end }} {{ end }} {{ end }} {{ with $dupes }}{{ $posts = complement . $posts }}{{ end }} {{ return $posts.ByDate.Reverse }}
